cmake_minimum_required(VERSION 3.22)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(frdm_mcxn947_robot_controller)
if(CONFIG_ROBOT_GELLO_UART AND CONFIG_ROBOT_GELLO_USB)
  message(FATAL_ERROR "Choose exactly one board GELLO transport: UART or USB")
endif()
if(CONFIG_ROBOT_GELLO_UART OR CONFIG_ROBOT_GELLO_USB)
  target_sources(app PRIVATE src/gello/gello_codec.c src/gello/gello_uart.c)
endif()
if(CONFIG_ROBOT_GELLO_USB)
  target_sources(app PRIVATE src/gello/gello_usb.c src/gello/gello_usb_desc.c src/gello/gello_usb_clock.c)
  target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/usb/host)
endif()
file(GLOB WL_SOURCES third_party/wirelink/src/*.c)
target_include_directories(app PRIVATE src src/generated third_party/wirelink/include)
target_sources(app PRIVATE src/main.c src/core/controller.c src/core/fault_ledger.c src/core/fault_manager.c src/core/clock_reference.c src/devices/can_codec.c src/devices/can_feedback.c src/devices/can_bus.c src/devices/can_trace.c src/generated/robot_messages.c ${WL_SOURCES})

if(CONFIG_ROBOT_PEER_LINK)
  target_sources(app PRIVATE src/peer/peer_codec.c src/peer/peer_link.c)
endif()

target_sources(app PRIVATE src/homing/gravity.c src/homing/home_path.c src/homing/home_still.c src/homing/paired_home.c src/homing/homing.c src/homing/local_home.c)
target_sources(app PRIVATE src/devices/stopped_can.c)
